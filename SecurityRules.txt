rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function admin() { return signedIn() && request.auth.token.role in ['ADMIN', 'SUPER_ADMIN']; }
    function owner(uid) { return signedIn() && request.auth.uid == uid; }
    function unchanged(field) { return !(field in request.resource.data.diff(resource.data).affectedKeys()); }
    function noServerFieldMutation() {
      return request.resource.data.diff(resource.data).affectedKeys().hasOnly([
        'firstname','lastname','displayName','photoURL','profile','preferences','lastLoginAt'
      ]);
    }

    // Roles, entitlements, billing state, verification state, and suspension are server-owned.
    match /users/{uid} {
      allow create: if owner(uid) && request.resource.data.keys().hasOnly([
        'userId','firstname','lastname','email','displayName','photoURL','profile','preferences','createdAt','lastLoginAt','authProvider'
      ]) && request.resource.data.userId == uid;
      allow read: if owner(uid) || admin();
      allow update: if owner(uid) && noServerFieldMutation();
      allow delete: if owner(uid) || admin();

      match /resumes/{resumeId} {
        allow read, create, update, delete: if owner(uid);
      }
      match /covers/{coverId} {
        allow read, create, update, delete: if owner(uid);
      }
      match /coverLetters/{coverId} {
        allow read, create, update, delete: if owner(uid);
      }
      match /favourites/{id} { allow read, create, update, delete: if owner(uid); }
      match /portfolios/{id} { allow read, create, update, delete: if owner(uid); }
      match /transactions/{id} { allow read: if owner(uid) || admin(); allow write: if false; }
    }

    match /portfolios/{id} {
      allow read: if resource.data.isPublished == true || (signedIn() && resource.data.userId == request.auth.uid) || admin();
      allow create: if signedIn() && request.resource.data.userId == request.auth.uid;
      allow update, delete: if signedIn() && resource.data.userId == request.auth.uid;
    }
    match /companies/{id} {
      allow read: if resource.data.status == 'approved' || admin() || (signedIn() && resource.data.employerId == request.auth.uid);
      allow create: if signedIn() && request.resource.data.employerId == request.auth.uid && request.auth.token.employer == true;
      allow update, delete: if admin() || (signedIn() && resource.data.employerId == request.auth.uid && request.resource.data.employerId == resource.data.employerId);
    }
    match /jobs/{id} {
      allow read: if true;
      allow create: if signedIn() && request.auth.token.employer == true && request.resource.data.employerId == request.auth.uid;
      allow update, delete: if admin() || (signedIn() && resource.data.employerId == request.auth.uid && request.resource.data.employerId == resource.data.employerId);
    }
    match /jobApplications/{id} {
      allow create: if signedIn() && request.resource.data.applicantUid == request.auth.uid;
      allow read: if admin() || (signedIn() && (resource.data.applicantUid == request.auth.uid || resource.data.employerId == request.auth.uid));
      allow update: if admin() || (signedIn() && resource.data.employerId == request.auth.uid && request.resource.data.applicantUid == resource.data.applicantUid);
      allow delete: if admin();
    }
    match /transactions/{id} { allow read: if signedIn() && resource.data.userId == request.auth.uid || admin(); allow write: if false; }
    match /subscriptions/{id} { allow read: if signedIn() && resource.data.userId == request.auth.uid || admin(); allow write: if false; }
    match /coupons/{id} { allow read: if signedIn(); allow write: if admin(); }
    match /blog_posts/{id} {
      allow read: if resource.data.status == 'approved' || (signedIn() && resource.data.authorUid == request.auth.uid) || admin();
      allow create: if signedIn() && request.resource.data.authorUid == request.auth.uid && request.resource.data.status == 'pending';
      allow update: if admin() || (signedIn() && resource.data.authorUid == request.auth.uid && resource.data.status in ['pending','rejected'] && request.resource.data.status == resource.data.status && request.resource.data.authorUid == resource.data.authorUid);
      allow delete: if admin();
    }
    match /blog_categories/{id} { allow read: if true; allow write: if admin(); }
    match /pages/{id} { allow read: if true; allow write: if admin(); }
    match /ads/{id} { allow read: if true; allow write: if admin(); }
    match /reviews/{id} { allow read: if true; allow write: if admin(); }
    match /trustedBy/{id} { allow read: if true; allow write: if admin(); }
    match /contact/{id} { allow create: if signedIn(); allow read, update, delete: if admin(); }
    match /notifications/{uid}/{document=**} { allow read, create, update, delete: if owner(uid); }
    match /data/{id} { allow read: if id in ['meta','system_settings','frontendstats']; allow write: if admin(); }
    match /settings/{id} { allow read, write: if admin(); }
    match /system_settings/{id} { allow read, write: if admin(); }
    match /{document=**} { allow read, write: if false; }
  }
}
