rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function admin() {
      return signedIn() && request.auth.token.role in ['ADMIN', 'SUPER_ADMIN'];
    }
    function employer() { return signedIn() && request.auth.token.employer == true; }
    function recentAuth() {
      return signedIn() && request.auth.token.auth_time is int
        && request.time.toMillis() - (request.auth.token.auth_time * 1000) < 600000;
    }
    function owner(uid) { return signedIn() && request.auth.uid == uid; }
    function ownEmail(email) {
      return signedIn() && request.auth.token.email is string && email == request.auth.token.email;
    }
    function changedOnly(fields) {
      return request.resource.data.diff(resource.data).affectedKeys().hasOnly(fields);
    }
    function jobOwner(jobId) {
      return signedIn() && get(/databases/$(database)/documents/jobs/$(jobId)).data.employerId == request.auth.uid;
    }

    // Identity, role, verification, suspension, billing and entitlements are server-owned.
    match /users/{uid} {
      allow create: if owner(uid)
        && request.resource.data.keys().hasOnly([
          'userId','firstname','lastname','email','displayName','photoURL','profile',
          'preferences','createdAt','lastLoginAt','authProvider','membership'
        ])
        && request.resource.data.userId == uid
        && ownEmail(request.resource.data.email)
        && (!('membership' in request.resource.data) || request.resource.data.membership == 'Basic');
      allow read: if owner(uid) || admin();
      allow update: if owner(uid) && changedOnly([
        'firstname','lastname','displayName','photoURL','profile','preferences','lastLoginAt','authProvider'
      ]);
      allow delete: if false;

      match /resumes/{resumeId} { allow read, create, update, delete: if owner(uid); }
      match /covers/{coverId} { allow read, create, update, delete: if owner(uid); }
      match /coverLetters/{coverId} { allow read, create, update, delete: if owner(uid); }
      match /favourites/{id} { allow read, create, update, delete: if owner(uid); }
      match /portfolios/{id} { allow read, create, update, delete: if owner(uid); }
      match /jobTracker/{id} { allow read, create, update, delete: if owner(uid); }
      match /invoices/{id} { allow read: if owner(uid) || admin(); allow write: if false; }
      match /transactions/{id} { allow read: if owner(uid) || admin(); allow write: if false; }
    }

    match /portfolios/{id} {
      allow read: if resource.data.isPublished == true
        || (signedIn() && resource.data.userId == request.auth.uid) || admin();
      allow create: if signedIn() && request.resource.data.userId == request.auth.uid;
      allow update: if admin() || (signedIn()
        && resource.data.userId == request.auth.uid
        && request.resource.data.userId == resource.data.userId);
      // Anonymous view counters may increment by exactly one and cannot alter content.
      allow update: if resource.data.isPublished == true
        && changedOnly(['views'])
        && request.resource.data.views == resource.data.views + 1;
      allow delete: if admin() || (signedIn() && resource.data.userId == request.auth.uid);
    }

    match /employerApplications/{uid} {
      allow create: if owner(uid)
        && request.resource.data.userId == request.auth.uid
        && request.resource.data.status == 'pending';
      allow read: if owner(uid) || admin();
      allow update: if owner(uid)
        && resource.data.userId == request.auth.uid
        && request.resource.data.userId == resource.data.userId
        && request.resource.data.status == 'pending';
      allow delete: if admin();
    }

    match /companies/{id} {
      allow read: if resource.data.status == 'approved' || admin()
        || (signedIn() && resource.data.employerId == request.auth.uid);
      allow create: if employer()
        && request.resource.data.employerId == request.auth.uid
        && request.resource.data.status == 'pending';
      allow update: if admin() || (employer()
        && resource.data.employerId == request.auth.uid
        && request.resource.data.employerId == resource.data.employerId
        && request.resource.data.status == resource.data.status);
      allow delete: if admin() || (employer() && resource.data.employerId == request.auth.uid);
    }

    match /jobs/{id} {
      allow read: if resource.data.status == 'active' || admin()
        || (signedIn() && resource.data.employerId == request.auth.uid);
      allow create: if employer()
        && request.resource.data.employerId == request.auth.uid
        && request.resource.data.status == 'pending'
        && request.resource.data.applicationsCount == 0;
      allow update: if admin() || (employer()
        && resource.data.employerId == request.auth.uid
        && request.resource.data.employerId == resource.data.employerId
        && request.resource.data.status == 'pending');
      // Applications are separate protected documents; this exception changes metrics only.
      allow update: if signedIn()
        && changedOnly(['applicationsCount','updatedAt'])
        && request.resource.data.applicationsCount == resource.data.applicationsCount + 1;
      allow delete: if admin() || (employer() && resource.data.employerId == request.auth.uid);
    }

    match /jobApplications/{id} {
      // Allows an applicant transaction to check its deterministic, ownership-bound
      // document before creation without granting collection/list access.
      allow get: if signedIn() && id.matches('^' + request.auth.uid + '_.*$');
      allow create: if signedIn()
        && id.matches('^' + request.auth.uid + '_.*$')
        && request.resource.data.userId == request.auth.uid
        && ownEmail(request.resource.data.applicantEmail)
        && request.resource.data.email == request.auth.token.email
        && request.resource.data.status == 'pending'
        && get(/databases/$(database)/documents/jobs/$(request.resource.data.jobId)).data.status == 'active';
      allow read: if admin()
        || (signedIn() && resource.data.userId == request.auth.uid)
        || jobOwner(resource.data.jobId);
      allow update: if admin() || (jobOwner(resource.data.jobId)
        && request.resource.data.userId == resource.data.userId
        && request.resource.data.jobId == resource.data.jobId
        && changedOnly(['status','statusUpdatedAt','employerNotes','updatedAt']));
      allow delete: if admin() || (signedIn()
        && resource.data.userId == request.auth.uid
        && resource.data.status == 'pending');
    }

    match /pb/{id} {
      allow read: if (resource.data.isPublished == true && resource.data.publicationMode == 'explicit')
        || (signedIn() && resource.data.ownerUid == request.auth.uid) || admin();
      allow create: if signedIn()
        && request.resource.data.ownerUid == request.auth.uid
        && request.resource.data.id == id;
      allow update, delete: if admin() || (signedIn()
        && resource.data.ownerUid == request.auth.uid
        && request.resource.data.ownerUid == resource.data.ownerUid);
    }

    match /transactions/{id} {
      allow read: if (signedIn() && resource.data.userId == request.auth.uid) || admin();
      allow write: if false;
    }
    match /subscriptions/{id} {
      allow read: if (signedIn() && resource.data.userId == request.auth.uid) || admin();
      allow write: if false;
    }
    match /payment_orders/{id} {
      allow read: if (signedIn() && resource.data.uid == request.auth.uid) || admin();
      allow write: if false;
    }
    match /invoices/{id} {
      allow read: if (signedIn() && resource.data.userId == request.auth.uid) || admin();
      allow write: if false;
    }
    match /coupons/{id} { allow read: if signedIn(); allow write: if admin(); }

    match /blog_posts/{id} {
      function validBlogFields(data) {
        return data.keys().hasOnly(['title', 'slug', 'content', 'excerpt', 'categoryId', 'authorUid', 'authorName', 'status', 'revision', 'seoTitle', 'seoDescription', 'createdAt', 'updatedAt', 'publishedAt', 'scheduledAt', 'viewCount', 'tags', 'featuredImage', 'adminNotes', 'rejectionReason', 'reviewedAt', 'reviewedBy'])
          && data.title is string && data.title.size() > 0 && data.title.size() <= 200
          && data.slug is string && data.slug.size() > 0 && data.slug.size() <= 180
          && data.content is string && data.content.size() <= 500000
          && data.excerpt is string && data.excerpt.size() <= 500
          && data.categoryId is string && data.categoryId.size() <= 128
          && data.authorUid is string && data.authorUid.size() > 0 && data.authorUid.size() <= 128
          && data.status in ['draft', 'pending', 'approved', 'rejected', 'scheduled']
          && data.revision is int && data.revision > 0
          && data.createdAt is timestamp && data.updatedAt is timestamp
          && data.viewCount is int && data.viewCount >= 0
          && (!data.keys().hasAny(['publishedAt']) || data.publishedAt == null || data.publishedAt is timestamp)
          && (!data.keys().hasAny(['scheduledAt']) || data.scheduledAt == null || data.scheduledAt is timestamp)
          && (!data.keys().hasAny(['seoTitle']) || (data.seoTitle is string && data.seoTitle.size() <= 120))
          && (!data.keys().hasAny(['seoDescription']) || (data.seoDescription is string && data.seoDescription.size() <= 320))
          && (!data.keys().hasAny(['authorName']) || (data.authorName is string && data.authorName.size() <= 160))
          && (!data.keys().hasAny(['adminNotes']) || (data.adminNotes is string && data.adminNotes.size() <= 2000))
          && (!data.keys().hasAny(['rejectionReason']) || (data.rejectionReason is string && data.rejectionReason.size() <= 1000))
          && (!data.keys().hasAny(['reviewedBy']) || (data.reviewedBy is string && data.reviewedBy.size() <= 128))
          && (!data.keys().hasAny(['reviewedAt']) || data.reviewedAt == null || data.reviewedAt is timestamp)
          && (!data.keys().hasAny(['tags']) || (data.tags is list && data.tags.size() <= 30))
          && (!data.keys().hasAny(['featuredImage']) || data.featuredImage == null || (data.featuredImage is string && data.featuredImage.size() <= 2048 && (data.featuredImage.matches('^https://[^\\s]+$') || data.featuredImage.matches('^/[^/].*$'))));
      }
      function currentRevision() {
        return resource.data.get('revision', 0);
      }
      function advancesRevision() {
        return request.resource.data.revision == currentRevision() + 1;
      }
      function memberFieldsOnly() {
        return request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          'title', 'slug', 'content', 'excerpt', 'categoryId', 'tags', 'featuredImage',
          'seoTitle', 'seoDescription', 'status', 'updatedAt', 'revision'
        ]);
      }
      function adminFieldsOnly() {
        return request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          'title', 'slug', 'content', 'excerpt', 'categoryId', 'tags', 'featuredImage',
          'seoTitle', 'seoDescription', 'status', 'scheduledAt', 'publishedAt',
          'updatedAt', 'revision', 'adminNotes', 'rejectionReason', 'reviewedAt', 'reviewedBy'
        ]);
      }
      function validAdminPublication() {
        return (request.resource.data.status == 'scheduled'
            && request.resource.data.scheduledAt is timestamp
            && request.resource.data.scheduledAt > request.time
            && request.resource.data.publishedAt == null)
          || (request.resource.data.status == 'approved'
            && request.resource.data.scheduledAt == null
            && request.resource.data.publishedAt is timestamp)
          || (request.resource.data.status in ['draft', 'pending', 'rejected']
            && request.resource.data.scheduledAt == null
            && request.resource.data.publishedAt == resource.data.publishedAt);
      }

      allow read: if resource.data.status == 'approved'
        || (signedIn() && resource.data.authorUid == request.auth.uid) || admin();
      allow create: if signedIn()
        && request.resource.data.authorUid == request.auth.uid
        && request.resource.data.status in ['draft', 'pending']
        && request.resource.data.revision == 1
        && request.resource.data.publishedAt == null
        && !request.resource.data.keys().hasAny(['scheduledAt', 'adminNotes', 'rejectionReason', 'reviewedAt', 'reviewedBy'])
        && validBlogFields(request.resource.data);
      allow update: if validBlogFields(request.resource.data)
        && request.resource.data.authorUid == resource.data.authorUid
        && request.resource.data.createdAt == resource.data.createdAt
        && request.resource.data.viewCount == resource.data.viewCount
        && advancesRevision()
        && ((admin() && adminFieldsOnly() && validAdminPublication())
          || (signedIn()
            && resource.data.authorUid == request.auth.uid
            && resource.data.status in ['draft', 'pending', 'rejected']
            && request.resource.data.status in ['draft', 'pending']
            && request.resource.data.publishedAt == resource.data.publishedAt
            && request.resource.data.get('scheduledAt', null) == resource.data.get('scheduledAt', null)
            && memberFieldsOnly()));
      allow delete: if admin() || (signedIn()
        && resource.data.authorUid == request.auth.uid
        && resource.data.status in ['draft', 'pending', 'rejected']);
    }
    match /blog_categories/{id} { allow read: if true; allow write: if admin(); }
    match /pages/{id} { allow read: if true; allow write: if admin(); }
    match /ads/{id} { allow read: if true; allow write: if admin(); }
    match /reviews/{id} { allow read: if resource.data.status == 'approved' || admin(); allow write: if admin(); }
    match /trustedBy/{id} { allow read: if true; allow write: if admin(); }
    match /contact/{id} { allow read, update, delete: if admin(); allow create: if false; }
    match /notifications/{uid}/{document=**} {
      allow read, update, delete: if owner(uid);
      allow create: if owner(uid);
    }

    // Public settings must live in the explicitly curated public_config document.
    // system_settings can contain provider credentials and is never client-public.
    match /data/{id} {
      allow read: if id in ['meta','frontendstats','public_config']
        || (admin() && !(id in ['system_settings','subscriptions']));
      allow write: if admin() && (!(id in ['system_settings','subscriptions']) || recentAuth());
    }
    match /settings/{id} {
      allow read: if admin() && !(id in ['ai_providers','payment_providers','oauth_providers']);
      allow write: if admin() && recentAuth();
    }
    match /system_settings/{id} { allow read: if admin(); allow write: if admin() && recentAuth(); }

    // Reset/verification tokens, provider secrets, audit logs and webhook ledgers are server-only.
    match /{document=**} { allow read, write: if false; }
  }
}
